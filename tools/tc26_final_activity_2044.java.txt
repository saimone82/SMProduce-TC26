package com.smproduce.palletsshipping;

import android.app.AlertDialog;
import android.content.Intent;
import android.graphics.Color;
import android.text.InputType;
import android.view.Gravity;
import android.widget.Button;
import android.widget.EditText;
import android.widget.LinearLayout;
import android.widget.TextView;

import java.util.Map;

/** Final TC26 launcher: cumulative Palletizing + Shipping + Modify Pallet + Scanner. */
public class TC26FinalActivity extends Tc26MainActivity {
    private static final String FINAL_VERSION = "2.0.44-tc26-final";

    @Override Map<String,String> map(String... values) {
        Map<String,String> params = super.map(values);
        params.put("app_version", FINAL_VERSION);
        return params;
    }

    @Override void buildShell() {
        super.buildShell();
        TextView badge = tv("SM PRODUCE TC26 · 2.0.44 FINAL", 14, Color.rgb(134,239,172));
        badge.setGravity(Gravity.CENTER);
        badge.setTypeface(null, 1);
        badge.setPadding(dp(8), dp(8), dp(8), dp(8));
        badge.setBackgroundColor(Color.rgb(19,32,51));
        root.addView(badge, 1, new LinearLayout.LayoutParams(-1, -2));
    }

    @Override void render() {
        super.render();
        if (step == Step.HOME) {
            TextView tools = tv(tr("TOOLS", "HERRAMIENTAS"), 13, Color.rgb(148,163,184));
            tools.setGravity(Gravity.CENTER);
            tools.setTypeface(null, 1);
            body.addView(tools);
            addSpace(6);

            Button scanner = choice("Scanner", "Escáner", v -> openScanner());
            scanner.setBackgroundColor(Color.rgb(52,65,85));

            Button modify = choice("Modify Existing Pallet", "Modificar pallet existente", v -> requestPalletEditPassword());
            modify.setBackgroundColor(Color.rgb(180,83,9));
        }
    }

    private void openScanner() {
        if (!online) {
            error(tr("Connect to the server before opening Scanner", "Conéctese al servidor antes de abrir el Escáner"));
            return;
        }
        final EditText input = new EditText(this);
        input.setInputType(InputType.TYPE_CLASS_NUMBER | InputType.TYPE_NUMBER_VARIATION_PASSWORD);
        final AlertDialog dialog = new AlertDialog.Builder(this)
                .setTitle(tr("Scanner password", "Contraseña del escáner"))
                .setMessage(tr("Enter the protected PIN", "Introduzca el PIN protegido"))
                .setView(input)
                .setNegativeButton(tr("Cancel", "Cancelar"), null)
                .setPositiveButton(tr("Open", "Abrir"), null)
                .create();
        dialog.setOnShowListener(x -> dialog.getButton(AlertDialog.BUTTON_POSITIVE).setOnClickListener(v -> {
            String pin = input.getText().toString().trim();
            if (pin.isEmpty()) return;
            call(map("action", "scanner_unlock", "password", pin), j -> {
                if (!dialog.isShowing()) return;
                dialog.dismiss();
                Intent intent = new Intent(this, ScannerActivity.class);
                intent.putExtra("scanner_password", pin);
                startActivity(intent);
            });
        }));
        dialog.show();
    }
}
